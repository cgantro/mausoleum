cmake_minimum_required(VERSION 3.20)
project(VoiceSafetyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
enable_testing()
add_executable(voice_safety_test voice_safety_test.cpp)
target_include_directories(voice_safety_test PRIVATE ../include ../../../A302/Source/A302Shared/Public)
target_link_libraries(voice_safety_test PRIVATE Threads::Threads)
add_test(NAME voice_safety COMMAND voice_safety_test)
if(MSVC)
  target_compile_options(voice_safety_test PRIVATE /UNDEBUG)
else()
  target_compile_options(voice_safety_test PRIVATE -UNDEBUG)
endif()
